\begin{frame}{Backup: exact derivatives on the contrast slice}
  \bookref{Chapter 3, Section 5}
  For
  \[
    h(u)=-u\log u-(1-u)\log(1-u),
  \]
  \[
    h''(u)=-\frac{1}{u(1-u)},
    \qquad
    h^{(4)}(u)=-\frac{2(1-3u+3u^2)}{u^3(1-u)^3}.
  \]
  At $u=1/2$,
  \[
    h''(1/2)=-4,
    \qquad
    h^{(4)}(1/2)=-32.
  \]
  Hence
  \[
    \mathcal G''_{\beta,J}(0)=2(4-2\beta J),
    \qquad
    \mathcal G^{(4)}_{\beta,J}(0)=64
      =24\left(\frac83\right).
  \]
\end{frame}

\begin{frame}{Backup: exact finite compatibility}
  \bookref{Chapter 5, Section 3}
  Suppose an infinite theory assigns a deficit $\mathcal I(G,H)$ to residual
  symmetry $H\leq G$. At minimum, finite compatibility asks for
  \[
    \mathcal I(G,H)=\log[G:H]
    \qquad\text{when }[G:H]<\infty.
  \]
  In the original setting,
  \[
    G=\Sym(X),\qquad H=\Aut(S),
  \]
  so
  \[
    \mathcal I(G,H)=\log\frac{|G|}{|H|}=I(S).
  \]

  Dimension, quotient measure, and entropy-rate proposals should therefore be
  viewed as category-specific continuations of the orbit-size principle.
\end{frame}
